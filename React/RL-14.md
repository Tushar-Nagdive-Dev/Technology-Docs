## 📄 Step 1: Create the Wrapper Component

📄 `FormWizard.jsx`

```jsx
import { useState } from 'react';
import Step1 from './Step1';
import Step2 from './Step2';
import Step3 from './Step3';

const steps = ['Step 1', 'Step 2', 'Step 3'];

export default function FormWizard() {
  const [step, setStep] = useState(0);
  const [formData, setFormData] = useState({
    fullName: '',
    email: '',
    password: '',
    confirmPassword: '',
    gender: '',
    terms: false
  });

  const next = () => setStep((s) => s + 1);
  const prev = () => setStep((s) => s - 1);

  const updateForm = (data) => setFormData((prev) => ({ ...prev, ...data }));

  const renderStep = () => {
    switch (step) {
      case 0:
        return <Step1 formData={formData} updateForm={updateForm} next={next} />;
      case 1:
        return <Step2 formData={formData} updateForm={updateForm} next={next} prev={prev} />;
      case 2:
        return <Step3 formData={formData} updateForm={updateForm} prev={prev} />;
      default:
        return <p>All steps done!</p>;
    }
  };

  return (
    <div>
      <h2>{steps[step]}</h2>
      {renderStep()}
    </div>
  );
}
```

## 📄 Step 2: Create Step Components

📄 `Step1.jsx`

```jsx
export default function Step1({ formData, updateForm, next }) {
  const { fullName, email } = formData;

  const handleNext = () => {
    if (!fullName || !email.includes('@')) return alert('Validation failed');
    next();
  };

  return (
    <div>
      <input
        value={fullName}
        onChange={(e) => updateForm({ fullName: e.target.value })}
        placeholder="Full Name"
      />
      <input
        value={email}
        onChange={(e) => updateForm({ email: e.target.value })}
        placeholder="Email"
      />
      <button onClick={handleNext}>Next</button>
    </div>
  );
}
```

📄 `Step2.jsx`

```jsx
export default function Step2({ formData, updateForm, next, prev }) {
  const { password, confirmPassword } = formData;

  const handleNext = () => {
    if (password.length < 6 || password !== confirmPassword) return alert('Password mismatch');
    next();
  };

  return (
    <div>
      <input
        type="password"
        value={password}
        onChange={(e) => updateForm({ password: e.target.value })}
        placeholder="Password"
      />
      <input
        type="password"
        value={confirmPassword}
        onChange={(e) => updateForm({ confirmPassword: e.target.value })}
        placeholder="Confirm Password"
      />
      <button onClick={prev}>Back</button>
      <button onClick={handleNext}>Next</button>
    </div>
  );
}
```

📄 `Step3.jsx`

```jsx
export default function Step3({ formData, updateForm, prev }) {
  const { gender, terms } = formData;

  const handleSubmit = () => {
    if (!gender || !terms) return alert('Please accept terms and select gender');
    alert('Form submitted successfully!');
  };

  return (
    <div>
      <select
        value={gender}
        onChange={(e) => updateForm({ gender: e.target.value })}
      >
        <option value="">Select Gender</option>
        <option value="male">Male</option>
        <option value="female">Female</option>
      </select>
      <label>
        <input
          type="checkbox"
          checked={terms}
          onChange={(e) => updateForm({ terms: e.target.checked })}
        />
        Accept Terms
      </label>
      <br />
      <button onClick={prev}>Back</button>
      <button onClick={handleSubmit}>Submit</button>
    </div>
  );
}
```
