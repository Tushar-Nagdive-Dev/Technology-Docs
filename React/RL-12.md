## 🧩 Part 2: Fetching Data with `fetch()`

```jsx
import { useEffect, useState } from 'react';

function PostList() {
  const [posts, setPosts] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState('');

  useEffect(() => {
    fetch('https://jsonplaceholder.typicode.com/posts?_limit=5')
      .then((res) => res.json())
      .then((data) => {
        setPosts(data);
        setLoading(false);
      })
      .catch((err) => {
        setError('Failed to fetch posts');
        setLoading(false);
      });
  }, []);

  if (loading) return <p>Loading posts...</p>;
  if (error) return <p>{error}</p>;

  return (
    <ul>
      {posts.map((post) => (
        <li key={post.id}>
          📌 <strong>{post.title}</strong>
        </li>
      ))}
    </ul>
  );
}
```

✅ Covers full flow: fetch → loading → success/error → render

## 🧩 Part 3: Using Axios (alternative to fetch)

```jsx
import axios from 'axios';
import { useEffect, useState } from 'react';

function PostList() {
  const [posts, setPosts] = useState([]);

  useEffect(() => {
    axios
      .get('https://jsonplaceholder.typicode.com/posts?_limit=5')
      .then((res) => setPosts(res.data));
  }, []);

  return (
    <ul>
      {posts.map((post) => (
        <li key={post.id}>{post.title}</li>
      ))}
    </ul>
  );
}
```

✅ Axios automatically parses JSON, handles errors with `.catch()`, and supports interceptors and base URLs.

## 🧩 Part 4: Adding Search Functionality

```jsx
function PostSearch() {
  const [query, setQuery] = useState('');
  const [posts, setPosts] = useState([]);

  useEffect(() => {
    if (query === '') return;

    const timer = setTimeout(() => {
      fetch(`https://jsonplaceholder.typicode.com/posts?q=${query}`)
        .then((res) => res.json())
        .then((data) => setPosts(data));
    }, 500); // debounce

    return () => clearTimeout(timer);
  }, [query]);

  return (
    <div>
      <input
        placeholder="Search posts"
        value={query}
        onChange={(e) => setQuery(e.target.value)}
      />
      <ul>
        {posts.map((p) => (
          <li key={p.id}>{p.title}</li>
        ))}
      </ul>
    </div>
  );
}
```

✅ Uses **debouncing** to limit API calls.

### 🧩 Part 5: Add Pagination (Manual Example)

```jsx
function PaginatedPosts() {
  const [page, setPage] = useState(1);
  const [posts, setPosts] = useState([]);

  useEffect(() => {
    fetch(`https://jsonplaceholder.typicode.com/posts?_page=${page}&_limit=5`)
      .then((res) => res.json())
      .then((data) => setPosts(data));
  }, [page]);

  return (
    <div>
      <button disabled={page <= 1} onClick={() => setPage(page - 1)}>Prev</button>
      <button onClick={() => setPage(page + 1)}>Next</button>
      <ul>
        {posts.map((p) => <li key={p.id}>{p.title}</li>)}
      </ul>
    </div>
  );
}
```

### ❗ Common Mistakes to Avoid

| Mistake                              | Solution                                |
| ------------------------------------ | --------------------------------------- |
| Infinite fetch loop                  | Add correct dependencies in `useEffect` |
| Not handling loading/errors          | Always manage all 3 states              |
| Multiple fetches for every keystroke | Use debouncing                          |
| Fetching on every render             | Wrap logic inside `useEffect`           |
